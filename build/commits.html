+def context_row(asof, panel, vol_window, index_col="INDEX"):
+    """One context row for `asof`, using ONLY panel rows at or before asof (PIT). Returns a plain dict so equality
+    is exact for the leakage test. `panel`: daily closes, columns = symbols (one is the index), index = dates."""
+    hist = panel.loc[:asof]                                  # PIT slice: nothing after asof is visible
+    idx = hist[index_col].astype(float)
+    logret = np.log(idx).diff()
+    # realised vol = std of the last vol_window index log-returns ending at asof (trailing window = PIT)
+    tail = logret.dropna().tail(int(vol_window))
+    rv = float(tail.std(ddof=0)) if len(tail) else 0.0

+    # breadth = fraction of NON-index symbols with a positive daily return on asof

+        up = (cur.astype(float) > prev.astype(float))
+        breadth = float(up.sum()) / float(len(syms))
+    else:
+        breadth = 0.0
+    return {"date": pd.Timestamp(asof).date().isoformat(),
+            "realized_vol": round(rv, 8), "breadth": round(breadth, 8), "index_ret": round(index_ret, 8)}
