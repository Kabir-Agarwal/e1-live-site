+"""NB19 HELD LOG — held orders as plain "held: <stock> — <reason>" lines for the BUILD ROOM's Messages page
+(lead-visible, never the public pages — NB9 removed held signals from public).
+
+The order path already records each held signal to reports/HELD-SIGNALS.log (TSV:
+`<ts>\\t[engine]\\t<stock>\\t<pattern>\\tHELD — <reason>`). This module is the plain, de-duplicated build-room
+VIEW of that log (`plain_lines`) plus an `append` for any caller that wants to write one directly. Pure/light —
+no heavy import. The build-room page rendering (build_journal) consumes plain_lines(); that render is the
+launch-path/site wire-in, LIVE via the site build.
+"""
+from __future__ import annotations
+
+import os
+
+_REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+LOG = os.path.join(_REPO, "reports", "HELD-SIGNALS.log")
+
+
+def append(stock, reason, path=None):
+    """Write one plain held line 'held: <stock> — <reason>'. Returns the line."""
+    path = path or LOG
+    line = "held: %s — %s" % (stock, reason)
+    with open(path, "a", encoding="utf-8") as f:
+        f.write(line + "\n")
+    return line
+
+
+def _reason(field):
+    """The plain reason from a HELD-SIGNALS field: the part after 'HELD — ', else the field itself."""
+    s = field.strip()
+    return s.split("HELD —", 1)[1].strip() if "HELD —" in s else (s[len("held:"):].strip()
+                                                                   if s.lower().startswith("held:") else s)
+
+
+def plain_lines(path=None):
+    """The build-room view: de-duplicated 'held: <stock> — <reason>' lines from the held log, order preserved.
+    Reads both the TSV order-path rows and any plain lines this module appended."""
+    path = path or LOG
+    out, seen = [], set()
+    try:
+        for ln in open(path, encoding="utf-8"):
+            ln = ln.rstrip("\n")
+            if not ln.strip():
+                continue
+            if "\t" in ln:                                   # order-path TSV: ts, [engine], stock, pattern, HELD—reason
+                parts = ln.split("\t")
+                if len(parts) >= 5:
+                    stock, reason = parts[2].strip(), _reason(parts[4])
+                else:
+                    continue
+            elif ln.lower().startswith("held:"):             # a plain line we appended
+                body = ln[len("held:"):].strip()
+                stock, _, reason = body.partition("—")
+                stock, reason = stock.strip(), reason.strip()
+            else:
+                continue
+            key = (stock, reason)
+            if key not in seen and stock:
+                seen.add(key)
+                out.append("held: %s — %s" % (stock, reason))
+    except OSError:
+        pass
+    return out
+

+        "2026-09-23T06:06:56+05:30\t[E10]\tAUBANK\tvwap|q4 …\tHELD — moment light not green (spread/depth)\n"

+"""NB19 item 53 — the 08:15 self-check line (replaces the 09:00 pre-open line).
+
+At 08:15 the machine checks itself and posts EXACTLY ONE line:
+  ✅ READY — N cameras up, gate open, N orders queued        (only when EVERY check passes)

+
+Pure over a status dict (testable, no heavy import). The runtime poster builds the status from vitals — each
+recorder lane's freshness (a row in the last 10 min, the status page's own measure), the REAL order gate
+(orders_allowed_now, NOT the kill switch) with its hold reason, and whether tomorrow's queue is loaded — then
+calls line(). READY is printed only after every check passes; a no-trade day is READY with "0 orders queued".
+The runtime wiring (vitals build → post at 08:15) is launch-path, LIVE via the nightly dry-run + restart.
+"""
+from __future__ import annotations
+
+CLOSER = "send this to the lead"          # NB21: a 🙋 (owner-action) line ends with a sanctioned action

+
+def line(status, expected_cameras=1):
+    """The 08:15 line for `status`. `status`: {streams:{name:{fresh:bool}}, orders_allowed_now, gates:{hold_reason},
+    queue_loaded:bool, orders_queued:int}. Every gate must pass for READY."""
+    streams = status.get("streams") or {}
+    up = [n for n, v in streams.items() if (v or {}).get("fresh")]
+    down = [n for n, v in streams.items() if not (v or {}).get("fresh")]

+    n_queued = int(status.get("orders_queued") or 0)

+        problems.append("order gate HELD (%s)" % reason)
+    if not queue_loaded:

+    if problems:
