+"""NB27 item 2 — the MORNING BAR RULE. So a late-published daily bar never costs a trading day again.
+
+The lesson from 2026-09-23: at 19:xx Kite had not yet published that day's daily bar, so the evening evaluator
+had only the previous day and could not build a valid next-open queue. But by the 06:00 login the next morning,
+Kite HAS published it. So the fix runs at login:
+
+  at login on day D:
+    prev = the last trading day before D
+    if prev's daily bar is NOT in the store  -> run the built top-up fetcher (Kite has it overnight) and re-check
+    if it is now present  -> run the conforming evaluator (write=True) on prev's close -> D's next-open queue
+    the 08:15 line then reports that queue (NONE fired stays a valid no-trade day)
+
+Pure decision + orchestration over injectable deps (calendar / store-latest / top-up / evaluate), so it is unit-
+tested without Kite, the store, or the evaluator. It RUNS at the 06:00 login (launch-path wire-in).
+"""
+from __future__ import annotations
+
+
+def run(today, prev_trading_day, store_latest_fn, topup_fn, evaluate_fn):
+    """Ensure prev's bar is in the store (top up if not), then evaluate prev's close for today's next-open queue.
+    Returns a dict describing what happened. All side-effecting deps are injected.
+      store_latest_fn() -> 'YYYY-MM-DD' latest bar date in the store (or '' if none)
+      topup_fn()        -> fetch the missing bar (the built Kite top-up); returns anything
+      evaluate_fn(asof) -> the conforming evaluator (write=True), returns the decisions list
+    """
+    latest = str(store_latest_fn() or "")
+    topped_up = False
+    if latest < prev_trading_day:                 # prev's bar not in the store yet -> fetch it (Kite has it now)
+        topup_fn()
+        topped_up = True
+        latest = str(store_latest_fn() or "")
+    if latest < prev_trading_day:                 # still missing after the top-up -> can't evaluate; no-trade day
+        return {"topped_up": topped_up, "bar_present": False, "queued": 0, "held": 0, "decisions": [],
+                "line": "🙋 morning — %s bar still missing after top-up (Kite not published?) — send this to the lead"
+                        % prev_trading_day}
+    decisions = evaluate_fn(prev_trading_day) or []
+    queued = [d for d in decisions if d.get("verdict") == "QUEUE"]
+    held = [d for d in decisions if d.get("verdict") == "HELD"]
+    return {"topped_up": topped_up, "bar_present": True, "queued": len(queued), "held": len(held),
+            "decisions": decisions,
+            "line": "✅ morning — %s bar in store%s, %d queued for %s"
+                    % (prev_trading_day, " (topped up)" if topped_up else "", len(queued), today)}
+
+
+def _demo():
+    # case A: prev's bar already present -> no top-up, evaluate
+    calls = {"topup": 0}
+    ev = lambda asof: [{"stock": "AUBANK", "verdict": "QUEUE"}, {"stock": "X", "verdict": "NO-FIRE"}]
+    r = run("2026-09-24", "2026-09-23", store_latest_fn=lambda: "2026-09-23",
+            topup_fn=lambda: calls.__setitem__("topup", calls["topup"] + 1), evaluate_fn=ev)
+    assert r["topped_up"] is False and r["bar_present"] and r["queued"] == 1 and calls["topup"] == 0, r
+    assert r["line"].startswith("✅ morning — 2026-09-23 bar in store,")
+    # case B: prev's bar missing, top-up lands it -> evaluate (topped up)
+    state = {"latest": "2026-09-22"}
+    def topup(): state["latest"] = "2026-09-23"
+    r = run("2026-09-24", "2026-09-23", store_latest_fn=lambda: state["latest"], topup_fn=topup, evaluate_fn=lambda a: [])
+    assert r["topped_up"] is True and r["bar_present"] and r["queued"] == 0 and "(topped up)" in r["line"], r
+    # case C: top-up fails to land it (Kite still hasn't published) -> no-trade day, 🙋
+    r = run("2026-09-24", "2026-09-23", store_latest_fn=lambda: "2026-09-22", topup_fn=lambda: None, evaluate_fn=lambda a: [])
+    assert r["bar_present"] is False and r["queued"] == 0 and r["line"].startswith("🙋 morning —") and "send this to the lead" in r["line"], r
+    print("morning_bar demo OK: bar present -> evaluate; missing -> top-up then evaluate; still missing -> 🙋 no-trade.")
+
