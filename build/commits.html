+The checker uses tracemalloc (Python-object growth), which is deterministic and immune to RSS noise. It is proven

+import tracemalloc

+from engine1_rebuild import book20_runner as B

+def growth_mb(run_fn, n):
+    """MB of Python-object growth over n calls of run_fn, after a warm-up + gc (measured per-poll leak signal)."""
+    tracemalloc.start()
+    try:
+        run_fn(); gc.collect()
+        base = tracemalloc.take_snapshot()
+        for _ in range(n):
+            run_fn()
+        gc.collect()
+        end = tracemalloc.take_snapshot()
+    finally:
+        tracemalloc.stop()
+    return sum(s.size_diff for s in end.compare_to(base, "lineno")) / (1024 * 1024)
+
