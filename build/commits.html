+"""NB43 #7 — real download files for the Data page, front-end only (§27e).
+
+For each recording lane the Data page shows, write a real CSV under the published site's downloads folder, from the
+FRONT-END recording view (site/data/recording.json) — never the desk's stores. The Data page's Excel button links
+to it. An empty lane still gets a file: the header + one plain "no rows yet" line — never a dead link.
+
+`slug(name)` is the shared key so the page row and its file always agree; `build(lanes, out_dir)` writes the files
+and returns {lane_name: "downloads/<slug>.csv"} for the page to wire. Pure/testable: `lanes` and the recording
+source are injectable; nothing here touches a desk store or a live process.
+"""
+from __future__ import annotations
+
+import json
+import os
+import re
+
+_REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+HEADER = ["date", "rows", "snapshots", "last_ts"]
+
+
+def slug(name):
+    return re.sub(r"[^a-z0-9]+", "-", str(name).lower()).strip("-") or "lane"
+
+
+def _recording_lanes(path=None):
+    try:
+        return (json.load(open(path or os.path.join(_REPO, "site", "data", "recording.json"),
+                                encoding="utf-8")).get("lanes")) or []
+    except Exception:
+        return []
+
+
+def _first_word(s):
+    return re.split(r"[ (/]", str(s).lower().strip(), 1)[0]
+
+
+def _history_for(lane_name, rec_lanes):
+    """The recording history rows for a page lane — matched to a recording.json lane by shared first word
+    ('Order-book depth' <-> 'Order-book depth (5 levels)'). [] if none."""
+    fw = _first_word(lane_name)
+    for rl in rec_lanes:
+        if _first_word(rl.get("lane", "")) == fw:
+            return rl.get("history") or []
+    return []
+
+
+def _csv_text(history):
+    lines = [",".join(HEADER)]
+    if history:
+        for h in history:
+            lines.append(",".join(str(h.get(k, "")) for k in HEADER))
+    else:
+        lines.append("no rows yet,,,")     # never a dead link: header + one honest line
+    return "\n".join(lines) + "\n"
+
+
+def build(lanes, out_dir, rec_path=None):
+    """Write one CSV per lane under out_dir; return {lane_name: 'downloads/<slug>.csv'}. `lanes` is a list of
+    lane-name strings (the page's lanes)."""
+    os.makedirs(out_dir, exist_ok=True)
+    rec = _recording_lanes(rec_path)
+    hrefs = {}
+    for name in lanes:
+        sg = slug(name)
+        with open(os.path.join(out_dir, sg + ".csv"), "w", encoding="utf-8") as f:
+            f.write(_csv_text(_history_for(name, rec)))
+        hrefs[name] = "downloads/%s.csv" % sg
+    return hrefs
+
+
+def _demo():
+    import tempfile
+    d = tempfile.mkdtemp()
+    rec = os.path.join(d, "recording.json")
+    json.dump({"lanes": [{"lane": "Order-book depth (5 levels)",
+                          "history": [{"date": "2026-09-22", "rows": 377016, "snapshots": 184, "last_ts": "..."}]}]},
+              open(rec, "w"))
+    out = os.path.join(d, "downloads")
+    hrefs = build(["Order-book depth", "Option chains"], out, rec_path=rec)
+    # matched lane -> real history rows; unmatched lane -> header + 'no rows yet'
+    ob = open(os.path.join(out, "order-book-depth.csv"), encoding="utf-8").read()
+    assert ob.splitlines()[0] == "date,rows,snapshots,last_ts" and "377016" in ob, ob
+    oc = open(os.path.join(out, "option-chains.csv"), encoding="utf-8").read()
+    assert oc.splitlines()[0] == "date,rows,snapshots,last_ts" and "no rows yet" in oc, oc
+    assert hrefs["Order-book depth"] == "downloads/order-book-depth.csv"
+    print("site_downloads demo OK: a real CSV per lane (matched -> history, empty -> header + 'no rows yet'); href map.")
+
+
+if __name__ == "__main__":
+    import sys
+    if "--demo" in sys.argv:
+        _demo()
