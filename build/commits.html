+import os, sys, tempfile
+sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

+def _isolate(monkey_running, held=False, token=[redacted]):
+    watchdog.STATE_FILE = os.path.join(tempfile.mkdtemp(), "wd_state.json")
+    watchdog.RESTART_LOG = os.path.join(tempfile.mkdtemp(), "restarts.log")
+    watchdog._held = lambda: held
+    watchdog._token_valid = lambda: token

+def _path(name):
+    safe = "".join(c if c.isalnum() or c in "-_" else "_" for c in str(name))
+    return os.path.join(LOCK_DIR, "%s.lock" % safe)

+def _pid_alive(pid):
+    try:
+        r = subprocess.run(["tasklist", "/FI", "PID eq %d" % int(pid)], capture_output=True, text=True, timeout=15)
+        return str(int(pid)) in (r.stdout or "")

+    """The live PID holding this lock, or None. A stale lock (dead PID / unreadable) reads as None."""
+    p = _path(name)
+    try:
+        pid = int(open(p, encoding="utf-8").read().strip() or "0")
+    except Exception:
+        return None

+    return None

+    h = holder(name)

+    should exit). Never blocks; a lock error never stops the only instance."""
+    try:
+        os.makedirs(LOCK_DIR, exist_ok=True)
+        h = holder(name)
+        if h and h != os.getpid():
