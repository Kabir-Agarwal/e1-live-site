+def event_band(values, lo_q, hi_q, min_history=20):
+    """Boolean event series: True at t where the ruler value sits within its own EXPANDING quantile band
+    [lo_q, hi_q] (quantiles computed from values up to and including t — PIT, never full-sample). Before
+    min_history observations the band is undefined -> False (fail-closed, no fabricated event)."""
+    v = pd.Series(values).astype(float)
+    lo = v.expanding(min_periods=min_history).quantile(lo_q)
+    hi = v.expanding(min_periods=min_history).quantile(hi_q)
+    return (v >= lo) & (v <= hi) & lo.notna() & hi.notna()
+
+
+def _outcome(closes, direction, horizon):
+    """1 when the forward `horizon`-day return goes the candidate's way, else 0. Daily cadence: entry next open is
+    approximated here by the close-to-close forward return (the fill law's exact next-open price is applied in the
+    paper loop, chunk 7)."""
+    fwd = closes.shift(-horizon) / closes - 1.0
+    return (np.sign(fwd) == np.sign(direction)).astype(float), fwd

+    """§5.4: the smallest n at which the Wilson half-interval width falls below `edge`. Half-width ~ z·sqrt(p(1-p)/n)
+    (widest at p=0.5, the conservative choice). Solve for n: n = (z^2 p(1-p)) / edge^2. Never a typed constant."""

+    return int(np.ceil((z ** 2) * p * (1 - p) / (edge ** 2)))

+A FORECAST ruler (here the HAR volatility forecast) must pass its own exam before it may sit on the shelf: fit on
+the chronological TRAIN split, score on TEST against the NAIVE forecast (yesterday's value / random walk). It

+chronological and the fit sees only TRAIN.

+import pandas as pd

+def har_features(rv, lags=(1, 5, 22)):
+    """Trailing HAR features: for each t, the mean realised vol over the last `lag` observations (inclusive of t).
+    All trailing => PIT (a feature at t never uses rv after t). Returns a DataFrame aligned to rv.index."""
+    d, w, m = lags
+    return pd.DataFrame({
+        "rv_d": rv.rolling(int(d)).mean(),
+        "rv_w": rv.rolling(int(w)).mean(),
+        "rv_m": rv.rolling(int(m)).mean(),
+    })
+
+
+def _fit_ols(X, y):
+    A = np.column_stack([np.ones(len(X)), X])
+    coef, *_ = np.linalg.lstsq(A, y, rcond=None)
+    return coef
+
+
+def _predict(coef, X):
+    return np.column_stack([np.ones(len(X)), X]) @ coef
+
+
+def exam(rv, train_frac=0.7, lags=(1, 5, 22)):
+    """Fit HAR on TRAIN, forecast next-day RV on TEST, compare to the naive (previous-value) forecast. Returns
+    har_rmse, naive_rmse, margin (naive-har mean squared-error gain), se (its standard error), beats_naive, and
+    decision FORECAST/STATE. beats_naive == margin > se (beats naive by more than one measured SE, §5.1)."""
+    feats = har_features(rv, lags)
+    target = rv.shift(-1)                                   # predict next day's RV
+    df = pd.concat([feats, target.rename("y"), rv.rename("naive")], axis=1).dropna()
+    n = len(df)
+    if n < 40:

+                "beats_naive": False, "decision": "STATE", "n": n}
+    cut = int(n * train_frac)
+    Xtr, ytr = df[["rv_d", "rv_w", "rv_m"]].iloc[:cut].values, df["y"].iloc[:cut].values
+    Xte, yte = df[["rv_d", "rv_w", "rv_m"]].iloc[cut:].values, df["y"].iloc[cut:].values
+    naive_te = df["naive"].iloc[cut:].values               # yesterday's value as the forecast
+    coef = _fit_ols(Xtr, ytr)                               # fit sees ONLY train (PIT)

+"""Claude Price Engine (engine 2) — CHUNK 1: the data + context layer (docs/architecture/CLAUDE-PRICE-ENGINE-
+ARCHITECTURE.md, C1).
+
+One context row per trading day — realised volatility, breadth, index return — computed from the SHARED daily-close
+panel READ-ONLY (§27a). Point-in-time by construction (§4.3): every quantity for date t is derived from
+`panel.loc[:t]` only, so a future bar can never change a past row. `vol_window` is a PARAMETER (no typed number,
+C2) chosen by walk-forward and recorded in the ruler record; this layer only computes given it.
+
+The panel is passed in (injectable) so the logic is unit-tested without loading the whole store; `build_from_store`
+wires the real datastore reader for the "regenerates from the stores" DONE criterion.
+"""
+from __future__ import annotations
+
+import numpy as np
+import pandas as pd
+
+
+def context_row(asof, panel, vol_window, index_col="INDEX"):
+    """One context row for `asof`, using ONLY panel rows at or before asof (PIT). Returns a plain dict so equality
+    is exact for the leakage test. `panel`: daily closes, columns = symbols (one is the index), index = dates."""
+    hist = panel.loc[:asof]                                  # PIT slice: nothing after asof is visible
+    idx = hist[index_col].astype(float)
+    logret = np.log(idx).diff()
+    # realised vol = std of the last vol_window index log-returns ending at asof (trailing window = PIT)
+    tail = logret.dropna().tail(int(vol_window))
+    rv = float(tail.std(ddof=0)) if len(tail) else 0.0
+    index_ret = float(logret.iloc[-1]) if len(logret) >= 1 and pd.notna(logret.iloc[-1]) else 0.0
+    # breadth = fraction of NON-index symbols with a positive daily return on asof
+    syms = [c for c in hist.columns if c != index_col]
+    if len(hist) >= 2 and syms:
+        prev, cur = hist[syms].iloc[-2], hist[syms].iloc[-1]
+        up = (cur.astype(float) > prev.astype(float))
+        breadth = float(up.sum()) / float(len(syms))
+    else:
+        breadth = 0.0
+    return {"date": pd.Timestamp(asof).date().isoformat(),
+            "realized_vol": round(rv, 8), "breadth": round(breadth, 8), "index_ret": round(index_ret, 8)}
+
+
+def build(panel, dates, vol_window, index_col="INDEX"):
+    """One context row per date in `dates`. DONE-criterion output — the context file the engine regenerates."""
+    return pd.DataFrame([context_row(d, panel, vol_window, index_col) for d in dates])
+

+    """Assemble the daily-close panel from the SHARED minute-bar store, READ-ONLY. `reader(token)` returns a
+    minute-bar frame (engine1_rebuild.datastore.read); we take the last close of each day. Heavy — used by
