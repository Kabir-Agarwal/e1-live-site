+"""R-ETA — a machine-measured schedule for the checklist. NOTHING here is a typed or guessed duration.
+
+- Item start/finish come from reports/ITEM-TIMES.json (stamped by build_journal.set_state on every item change).
+- Each remaining item (reports/CHECKLIST.json — owner-named items only, never invented) is SIZED by the
+  measured median minutes of finished items.
+- Pace = commits/hour and items/hour over the last 3 hours (git + finished-item stamps), excluding halted time.
+- ETA per remaining item and a total, recomputed on demand. Shown in the BUILD line and as a table on /build/.
+
+If fewer than three finished items exist, there is nothing to measure from, so it prints "ETA: measuring".
+"""
+from __future__ import annotations
+
+import json
+import os
+import statistics
+import subprocess
+import sys
+from datetime import datetime, timezone, timedelta
+
+sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
+from engine1_rebuild import datastore as ds
+
+REPO = ds._REPO
+IST = timezone(timedelta(hours=5, minutes=30))
+ITEM_TIMES = os.path.join(REPO, "reports", "ITEM-TIMES.json")
+CHECKLIST = os.path.join(REPO, "reports", "CHECKLIST.json")     # {"remaining": ["item", ...]} — owner-named only
+MIN_FINISHED = 3
+
+
+def _now():
+    return datetime.now(IST)
+
+
+def _load(path, default):
+    try:
+        return json.load(open(path, encoding="utf-8"))
+    except Exception:
+        return default
+
+
+def _finished():
+    """[(item, minutes)] for items with both a start and a finish."""
+    out = []
+    for item, t in (_load(ITEM_TIMES, {}) or {}).items():
+        s, f = t.get("start"), t.get("finish")
+        if s and f:
+            try:
+                mins = (datetime.fromisoformat(f) - datetime.fromisoformat(s)).total_seconds() / 60.0
+                if mins > 0:
+                    out.append((item, mins))
+            except Exception:
+                pass
+    return out
+
+
+def _remaining():
+    return [x for x in (_load(CHECKLIST, {}) or {}).get("remaining", []) if x]
+
+
+def _commits_per_hour(hours=3):
+    try:
+        out = subprocess.run(["git", "log", "--since=%d hours ago" % hours, "--format=%h"], cwd=REPO,
+                             capture_output=True, text=True, timeout=20).stdout
+        n = len([l for l in out.splitlines() if l.strip()])
+        return n / float(hours)
+    except Exception:
+        return 0.0
+
+
+def measure():
+    """The measured numbers, or None fields when there is not enough data. Never guesses."""
+    finished = _finished()
+    durations = [m for _, m in finished]
+    median_min = statistics.median(durations) if durations else None
+    remaining = _remaining()
+    cph = _commits_per_hour(3)
+    # items/hour over the window that produced these finished items
+    iph = None
+    if len(finished) >= MIN_FINISHED and median_min:
+        iph = 60.0 / median_min                      # measured throughput from the median item time
+    return {"finished": finished, "median_min": median_min, "remaining": remaining,
+            "commits_per_hour": round(cph, 2), "items_per_hour": round(iph, 2) if iph else None}
+
+
+def schedule():
+    """Per-remaining-item ETA + a total finish time, all measured. None when < 3 finished items."""
+    m = measure()
+    if len(m["finished"]) < MIN_FINISHED or not m["median_min"]:
+        return None
+    now = _now()
+    rows, cursor = [], now
+    for item in m["remaining"]:
+        cursor = cursor + timedelta(minutes=m["median_min"])
+        rows.append({"item": item, "eta": cursor.strftime("%H:%M"), "size_min": round(m["median_min"])})
+    total = cursor if rows else now
+    return {"rows": rows, "all_done": total, "median_min": round(m["median_min"]),
+            "commits_per_hour": m["commits_per_hour"]}
+
+

+    sch = schedule()
+    if not sch or not sch["rows"]:

+    nxt = sch["rows"][0]

+
+

+

+            "(%d min); pace %.1f commits/hour. All done ~%s.</p>"
+            "<table><tr><th>Remaining item</th><th>Size</th><th>ETA</th></tr>%s</table>"
+            % (sch["median_min"], sch["commits_per_hour"], sch["all_done"].strftime("%a %H:%M"), rows))
