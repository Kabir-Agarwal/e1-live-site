+"""NB40 item 3 — the twin-run REHEARSAL ROOT.
+
+Redirect every LIVE store/ledger/notify path the order path writes, so a replay of a recorded day cannot touch a
+live file. Built + tested here; the actual Tuesday replay runs after close in the isolated clone (twin_run).
+
+Two independent layers, matching twin_run's design, so a miss in either still fails closed:
+  PREVENTIVE — `redirect(root)` monkeypatches the known write-target module constants (paper.LEDGER,
+    book20_runner.LEDGER, book20_runner.READOUT) to files UNDER `root`, and swaps notify.notify for a capture that
+    posts nothing, for the duration of the replay; everything is restored on exit.
+  DETECTIVE  — `no_live_writes(fn)` hashes every live ledger/store (twin_run's guarded set, which since NB38
+    includes the MAIN paper ledger) before and after fn, and reports any path that changed. A change is the loud
+    failure — so even a write path `redirect` forgot is caught.
+
+Nothing here runs the heavy live loop; it is the safe harness the twin run uses at run time.
+"""
+from __future__ import annotations
+
+import contextlib
+import hashlib
+import os
+
+_REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+
+
+def _write_targets():
+    """(module, attr) for every module CONSTANT that names a live write path the order path touches. Each is a
+    plain path string used via `open(<CONST>, ...)`, so re-pointing the constant re-points the write."""
+    from engine1_rebuild import paper
+    from engine1_rebuild import book20_runner as B
+    return [(paper, "LEDGER"), (B, "LEDGER"), (B, "READOUT")]
+
+
+@contextlib.contextmanager
+def redirect(root):
+    """PREVENTIVE: point every live write-target under `root` and silence notify, for the duration. Restores on
+    exit even if the body raises. Yields {'root', 'posts', 'targets'} — posts collects what notify WOULD have sent."""
+    os.makedirs(root, exist_ok=True)
+    saved, targets = [], {}
+    for mod, attr in _write_targets():
+        old = getattr(mod, attr)
+        saved.append((mod, attr, old))
+        new = os.path.join(root, os.path.basename(str(old)))
+        setattr(mod, attr, new)
+        targets["%s.%s" % (mod.__name__, attr)] = new
+    from scripts import notify as _n
+    old_notify, posts = _n.notify, []
+    _n.notify = lambda *a, **k: posts.append(a[0] if a else "")
+    try:
+        yield {"root": root, "posts": posts, "targets": targets}
+    finally:
+        for mod, attr, old in saved:
+            setattr(mod, attr, old)
+        _n.notify = old_notify

+def live_paths():

+def _hash(paths):
+    out = {}
+    for p in paths:
+        try:
+            out[p] = hashlib.sha256(open(p, "rb").read()).hexdigest()
+        except Exception:
+            out[p] = None
+    return out

+
+def no_live_writes(fn, paths=None):

+    fn()
+    after = _hash(paths)
+    changed = sorted(p for p in (set(before) | set(after)) if before.get(p) != after.get(p))
+    return (not changed), changed
+
