+"""SessionEnd hook (CLAUDE.md §14.1 + §23.3) — the exit half of the single-instance marker.
+
+The watchdog's auto-continue (session_resurrect._session_active) treats a session as LIVE while
+scheduler/.owner_active is fresh (< 4h). The SessionStart hook writes it; nothing cleared it on exit, so after a
+real session end the marker stayed "fresh" for up to 4h and the watchdog never reopened (the NB32/NB33 gap). This
+hook CLEARS the marker when the session actually ends, so the watchdog can reopen the next item (RAM + pending +
+not-login permitting). It fires on a real SessionEnd, never on an idle turn, so a live-but-idle session is never
+double-spawned. Fail-open: any error is swallowed.
+"""
+import os
+import sys
+
+REPO = os.environ.get("E1_REPO") or r"[redacted]\Desktop\trading-engines-v2"
+
+
+def clear_marker(path=None):
+    p = path or os.path.join(REPO, "scheduler", ".owner_active")
+    try:
+        if os.path.exists(p):
+            os.remove(p)
+        return True
+    except Exception:
+        return False
+
+
+def main():
+    try:
+        sys.stdin.read()          # consume the hook payload (reason etc.); we clear on any real end
+    except Exception:
+        pass
+    clear_marker()
+    sys.exit(0)
+
+
+if __name__ == "__main__":
+    main()
