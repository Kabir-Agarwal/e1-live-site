+"""NB25 item 3 — the EVENING PRE-CHECK. Before the evaluator runs each evening, one light line:
+  ✅ evening — today's bar in store, depth rows today N, RAM ok
+  🙋 evening — <gap, plain words> — send this to the lead
+
+The lesson from 2026-09-23: the evaluator ran on a STALE bar (store latest was Tuesday; Wednesday never ingested)
+and re-produced already-expired signals. This gate stops that: it passes ONLY when today's bar is actually in the
+store, the depth recorder has rows today, and RAM is above the floor. Pure over a status dict (testable, no heavy
+import); the runtime fills it from the store + the depth stream + memory_guard, and posts line() with the machine
+clock. 🙋 ends with the sanctioned closer; ✅ never asks the owner to do anything.
+"""
+from __future__ import annotations
+
+CLOSER = "send this to the lead"
+
+
+def line(status):
+    """status: {bar_date, today, depth_rows_today:int, ram_ok:bool}. ✅ only if every check passes."""
+    bar_date = str(status.get("bar_date") or "")
+    today = str(status.get("today") or "")
+    depth = int(status.get("depth_rows_today") or 0)
+    ram_ok = bool(status.get("ram_ok"))
+    problems = []
+    if bar_date != today:
+        problems.append("today's bar (%s) not in store — latest is %s" % (today or "?", bar_date or "none"))
+    if depth <= 0:
+        problems.append("no depth rows recorded today")
+    if not ram_ok:
+        problems.append("free RAM below the floor")
+    if problems:
+        return "🙋 evening — %s — %s" % ("; ".join(problems), CLOSER)
+    return "✅ evening — today's bar in store, depth rows today %d, RAM ok" % depth
+
+
+def _demo():
+    ok = {"bar_date": "2026-09-24", "today": "2026-09-24", "depth_rows_today": 5000, "ram_ok": True}
+    assert line(ok) == "✅ evening — today's bar in store, depth rows today 5000, RAM ok", line(ok)
+    # tonight's exact failure: store latest is Tuesday, evaluator asked to run on Wednesday
+    stale = {"bar_date": "2026-09-22", "today": "2026-09-23", "depth_rows_today": 0, "ram_ok": True}
+    r = line(stale)
+    assert r.startswith("🙋 evening —") and "not in store" in r and "no depth rows" in r and r.endswith(CLOSER), r
+    lowram = {"bar_date": "2026-09-24", "today": "2026-09-24", "depth_rows_today": 10, "ram_ok": False}
+    assert "RAM below the floor" in line(lowram)
+    print("evening_precheck demo OK: ✅ only when today's bar + depth rows + RAM; 🙋 names the gap (the Wed lesson).")
+
+
+if __name__ == "__main__":
+    import sys
+    if "--demo" in sys.argv:
+        _demo()

+"""NB19 HELD LOG — held orders as plain "held: <stock> — <reason>" lines for the BUILD ROOM's Messages page
+(lead-visible, never the public pages — NB9 removed held signals from public).
+
+The order path already records each held signal to reports/HELD-SIGNALS.log (TSV:
+`<ts>\\t[engine]\\t<stock>\\t<pattern>\\tHELD — <reason>`). This module is the plain, de-duplicated build-room
+VIEW of that log (`plain_lines`) plus an `append` for any caller that wants to write one directly. Pure/light —
+no heavy import. The build-room page rendering (build_journal) consumes plain_lines(); that render is the
+launch-path/site wire-in, LIVE via the site build.
+"""
+from __future__ import annotations
+
+import os
+
+_REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+LOG = os.path.join(_REPO, "reports", "HELD-SIGNALS.log")
+
+
+def append(stock, reason, path=None):
+    """Write one plain held line 'held: <stock> — <reason>'. Returns the line."""
+    path = path or LOG
+    line = "held: %s — %s" % (stock, reason)
+    with open(path, "a", encoding="utf-8") as f:
+        f.write(line + "\n")
+    return line
+
+
+def _reason(field):
+    """The plain reason from a HELD-SIGNALS field: the part after 'HELD — ', else the field itself."""
+    s = field.strip()
+    return s.split("HELD —", 1)[1].strip() if "HELD —" in s else (s[len("held:"):].strip()
+                                                                   if s.lower().startswith("held:") else s)
+
+
+def plain_lines(path=None):
+    """The build-room view: de-duplicated 'held: <stock> — <reason>' lines from the held log, order preserved.
+    Reads both the TSV order-path rows and any plain lines this module appended."""
+    path = path or LOG
+    out, seen = [], set()
+    try:
+        for ln in open(path, encoding="utf-8"):
+            ln = ln.rstrip("\n")
+            if not ln.strip():
+                continue
+            if "\t" in ln:                                   # order-path TSV: ts, [engine], stock, pattern, HELD—reason
+                parts = ln.split("\t")
+                if len(parts) >= 5:
+                    stock, reason = parts[2].strip(), _reason(parts[4])
+                else:
+                    continue
+            elif ln.lower().startswith("held:"):             # a plain line we appended
+                body = ln[len("held:"):].strip()
+                stock, _, reason = body.partition("—")
+                stock, reason = stock.strip(), reason.strip()
+            else:
+                continue
+            key = (stock, reason)
+            if key not in seen and stock:
+                seen.add(key)
+                out.append("held: %s — %s" % (stock, reason))
+    except OSError:
+        pass
+    return out
+
