+"""PRICE ENGINE V2 — FLOOR chunk 2: the judge (§2.6 / §2.19).
+
+Trust score T: T0 = 1; each trading day after close T <- T * (1 + r_t), where r_t is the day's net rupee result of
+the pattern's combined paper bet / rupees deployed (all that day's fills = one bet, one move; no trade -> r_t = 0).
+Promotion bar B = N_A / α (N_A = admitted Door-A patterns, near-copies merged; α the shared luck allowance). A
+pattern is promoted when T >= B AND the product of (1 + r_t) over the last W days >= 1 (lifetime AND recent
+window, §2.8/§2.19).
+
+Why it is safe to check nightly (Ville's inequality): for a pattern with no edge after costs, {T_t} is a
+non-negative super-martingale with T0 = 1, so P(sup_t T_t >= B) <= 1/B, however often it is checked. Per pattern
+1/B = α/N_A; summed over N_A patterns the expected false promotions <= α.
+
+α is a DECLARED CONVENTION, never a bare typed number (§2.19 / §4.1):
+  ALPHA = 0.05 — the standard family-wise false-promotion allowance: across ALL admitted patterns the expected
+  number of worthless ones ever promoted is <= 0.05. Sensitivity: halving α to 0.025 doubles B and roughly
+  doubles the days a real edge needs to cross it; the owner may tighten α (never loosen it without a decision).
+"""
+from __future__ import annotations
+
+import numpy as np
+
+ALPHA = 0.05                 # declared convention (see module docstring) — the shared luck allowance
+
+
+def trust_update(T, r_t):
+    return T * (1.0 + r_t)
+
+
+def promotion_bar(n_admitted, alpha=ALPHA):
+    """B = N_A / α (§2.19). n_admitted counts admitted patterns with near-copies merged, not the whole family."""
+    return float(n_admitted) / float(alpha)
+
+
+def is_promoted(T, B, recent_returns):
+    """Lifetime AND recent-window both positive: T >= B AND product(1 + r_t over the last W) >= 1 (§2.8)."""
+    if T < B:
+        return False
+    prod = 1.0
+    for r in recent_returns:
+        prod *= (1.0 + r)
+    return prod >= 1.0
+
+
+def monte_carlo(n_streams, days, daily_edge, g, n_admitted, alpha=ALPHA, seed=0):
+    """Fake-data test. Each stream: r_t = daily_edge ± g fair coin (worthless => daily_edge 0 => E[r]=0 martingale;
+    real edge => positive drift). Tracks whether T EVER crosses B over the stream's life, and the first crossing
+    day. Returns promoted_frac (fraction of streams that cross B) and median_days_to_cross (over the crossers)."""
+    rng = np.random.default_rng(seed)
+    B = promotion_bar(n_admitted, alpha)
+    signs = rng.integers(0, 2, size=(n_streams, days)) * 2 - 1        # ±1
+    r = daily_edge + g * signs                                        # daily returns
+    logT = np.cumsum(np.log1p(r), axis=1)                             # log trust path
+    crossed_mask = logT >= np.log(B)
+    ever = crossed_mask.any(axis=1)
+    promoted_frac = float(ever.mean())
+    first = np.argmax(crossed_mask, axis=1)                           # first True per row (0 if none, filtered by ever)
+    cross_days = first[ever]
+    median_days = int(np.median(cross_days)) + 1 if len(cross_days) else 0   # +1: day index -> day count
+    return {"promoted_frac": promoted_frac, "median_days_to_cross": median_days, "bar": B, "alpha": alpha}
+
+
+def merge_near_copies(names, corr, threshold):
+    """Union-find grouping: patterns with |corr| >= threshold are near-copies and count as ONE. Returns the number
+    of merged groups (= N_A contribution). corr is a symmetric matrix aligned to names."""
+    n = len(names)
+    parent = list(range(n))
+
+    def find(x):
+        while parent[x] != x:
+            parent[x] = parent[parent[x]]; x = parent[x]
+        return x
+
+    for i in range(n):
+        for j in range(i + 1, n):
+            if abs(corr[i][j]) >= threshold:
+                parent[find(i)] = find(j)
+    return len({find(i) for i in range(n)})
+
+

+    w = monte_carlo(2000, 1200, 0.0, 0.01, 50, 0.05, seed=1)

+
