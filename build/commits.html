+"""Claude Price Engine (engine 2) — CHUNK 1 per §9: ruler shelf v1 on the bar store + sanity report.
+
+Computes STATE rulers on a daily OHLCV frame per symbol, READ-ONLY off the shared store (§27a). Families A (price/
+trend) and C (volatility) are wired here as shelf v1; families B/E/F/G/H register the same way as their §4 data
+gates open — a family with no data is simply ABSENT from the row, never placeholdered (§4). Every ruler is
+point-in-time (§4.3): its value at `asof` is computed from `bars.loc[:asof]` only, so a future bar never changes a
+past value. Windows are BANDS passed in (§3) — no typed number lives in this module.
+
+The pure per-symbol computation is unit-tested; `compute_panel` + `sanity_report` are the §9 chunk-1 deliverable
+(coverage / NaN / range per ruler), and `build_from_store` wires the real datastore reader for the full-store run.
+"""
+from __future__ import annotations
+
+import numpy as np
+import pandas as pd
+
+
+# --- individual STATE rulers (pure, trailing = PIT) -------------------------------------------------------------
+def _rsi(close, n):
+    d = close.diff()
+    up = d.clip(lower=0).ewm(alpha=1 / n, adjust=False).mean()
+    dn = (-d.clip(upper=0)).ewm(alpha=1 / n, adjust=False).mean()
+    rs = up / dn.replace(0, np.nan)
+    return (100 - 100 / (1 + rs)).iloc[-1]
+
+
+def _realized_vol(close, n):
+    r = np.log(close).diff().dropna().tail(int(n))
+    return float(r.std(ddof=0)) if len(r) else np.nan
+
+
+def _boll_z(close, n):
+    tail = close.tail(int(n))
+    sd = tail.std(ddof=0)
+    return float((close.iloc[-1] - tail.mean()) / sd) if sd and not np.isnan(sd) else np.nan
+
+
+def _atr(bars, n):
+    h, l, c = bars["high"], bars["low"], bars["close"]
+    pc = c.shift(1)
+    tr = pd.concat([(h - l), (h - pc).abs(), (l - pc).abs()], axis=1).max(axis=1)
+    return float(tr.tail(int(n)).mean())
+
+
+def _gap_pct(bars):
+    if len(bars) < 2:
+        return np.nan
+    return float(bars["open"].iloc[-1] / bars["close"].iloc[-2] - 1.0)
+
+
+# registry: ruler name -> (fn(bars, asof-slice, windows) -> value). Families B/E/F/G/H append here as gates open.
+def _registry():
+    return {
+        "rsi":          lambda b, w: _rsi(b["close"], w.get("rsi", 14)),
+        "realized_vol": lambda b, w: _realized_vol(b["close"], w.get("vol", 20)),
+        "boll_z":       lambda b, w: _boll_z(b["close"], w.get("boll", 20)),
+        "atr":          lambda b, w: _atr(b, w.get("atr", 14)),
+        "gap_pct":      lambda b, w: _gap_pct(b),
+    }
+
+
+def compute_row(bars, asof, windows):
+    """Every registered ruler for one symbol at `asof`, PIT (only bars at/before asof). Empty bars -> {} (a ruler
+    with no data is ABSENT, §4). A ruler that returns NaN is kept as NaN (the sanity report counts it); a ruler
+    that RAISES is absent for this symbol (never a fabricated value)."""
+    b = bars if asof is None else bars.loc[:asof]
+    if not len(b):
+        return {}
+    out = {}
+    for name, fn in _registry().items():
+        try:
+            out[name] = fn(b, windows)
+        except Exception:
+            continue                      # absent, never placeholdered
+    return out
+
+
+def compute_panel(bars_by_symbol, dates, windows):
+    """One row per (symbol, date). The shelf-v1 table the sanity report summarises."""
+    rows = []
+    for sym, bars in bars_by_symbol.items():
+        for d in dates:
+            r = compute_row(bars, d, windows)
+            if r:
+                rows.append({"symbol": sym, "date": pd.Timestamp(d).date().isoformat(), **r})
+    return pd.DataFrame(rows)
+
+
+def sanity_report(panel_df):
+    """§9 chunk-1 deliverable: per ruler, coverage (non-NaN fraction), NaN count, and value range."""
+    rep = {}
+    ruler_cols = [c for c in panel_df.columns if c not in ("symbol", "date")]
+    n = len(panel_df)
+    for c in ruler_cols:
+        col = pd.to_numeric(panel_df[c], errors="coerce")
+        nn = int(col.notna().sum())
+        rep[c] = {"coverage": round(nn / n, 4) if n else 0.0, "n_nan": int(n - nn),
+                  "min": (None if nn == 0 else float(col.min())), "max": (None if nn == 0 else float(col.max()))}
+    return rep
+
+
+def build_from_store(reader, tokens, dates, windows):
+    """Wire the real 1-min store (engine1_rebuild.datastore.read) → daily OHLCV per token → the shelf panel.
+    READ-ONLY. Heavy (used for the full-store §9 run, not in unit tests)."""
+    bars = {}
+    for tok in tokens:
+        df = reader(tok)
+        if df is None or not len(df):
+            continue
+        daily = pd.DataFrame({
+            "open": df["open"].resample("1D").first(), "high": df["high"].resample("1D").max(),
+            "low": df["low"].resample("1D").min(), "close": df["close"].resample("1D").last(),
+            "volume": df["volume"].resample("1D").sum()}).dropna(how="all")
+        bars[str(tok)] = daily
+    return compute_panel(bars, dates, windows)
+
+

+          "(coverage/NaN/range).")
+

+    import sys

+        _demo()
