+pulls the TOP unchecked backlog item into the checklist and logs it, so autonomous progress never stalls while

+    return os.environ.get("E1_BACKLOG") or os.path.join(REPO, "docs", "BACKLOG.md")

+    return os.environ.get("E1_CHECKLIST") or os.path.join(REPO, "reports", "CHECKLIST.json")

+_LINE = re.compile(r"^- \[( |~|x)\] (.+?)\s*$")
+
+
+def items():
+    """[(state, text)] in file order; state is ' ' (queued), '~' (active) or 'x' (done)."""
+    out = []
+    try:
+        for ln in open(_bl_path(), encoding="utf-8"):
+            m = _LINE.match(ln.rstrip("\n"))
+            if m:
+                out.append((m.group(1), m.group(2)))
+    except Exception:
+        pass
+    return out
+
+
+def next_item():
+    for state, text in items():
+        if state == " ":
+            return text
+    return None
+
+
+def is_empty():
+    return next_item() is None
+
+
+def _checklist():
+    try:
+        return json.load(open(_cl_path(), encoding="utf-8"))
+    except Exception:
+        return {"remaining": [], "done": []}
+
+
+def pull_next(logger=None):
+    """If the checklist is empty of remaining items, move the top queued backlog item into it and mark that
+    backlog line active. Returns the pulled item, or None. Logs the pull."""
+    c = _checklist()
+    if [x for x in (c.get("remaining") or []) if x]:
+        return None                                   # checklist still has work — nothing to pull
+    nxt = next_item()
+    if not nxt:
+        return None
+    # mark the backlog line active
+    try:
+        lines = open(_bl_path(), encoding="utf-8").read().splitlines()
+        for i, ln in enumerate(lines):
+            m = _LINE.match(ln)
+            if m and m.group(1) == " " and m.group(2) == nxt:
+                lines[i] = ln.replace("- [ ]", "- [~]", 1)
+                break
+        open(_bl_path(), "w", encoding="utf-8").write("\n".join(lines) + "\n")
+    except Exception:
+        pass
+    c.setdefault("remaining", []).append(nxt)
+    try:
+        json.dump(c, open(_cl_path(), "w", encoding="utf-8"), indent=1)
+    except Exception:
+        pass
+    line = "%s pulled backlog -> checklist: %s" % (datetime.now(IST).strftime("%Y-%m-%d %H:%M"), nxt)
+    try:
+        os.makedirs(os.path.dirname(LOG), exist_ok=True)
+        open(LOG, "a", encoding="utf-8").write(line + "\n")
+    except Exception:
+        pass
+    if logger:
+        try:
+            logger(line)
+        except Exception:
+            pass
+    return nxt
+
+
+def all_empty():
+    """True when neither the checklist nor the backlog has any remaining work — the IDLE state."""
+    c = _checklist()
+    return not [x for x in (c.get("remaining") or []) if x] and is_empty()

+def _find_block(s, tag, start=0):
+    """Find the next <tag …>…</tag> from `start`, honouring nesting of the same tag. Returns
+    (open_start, inner_start, inner_end, close_end, attrs) or None."""
+    om = re.compile(r"<%s\b([^>]*)>" % tag)
