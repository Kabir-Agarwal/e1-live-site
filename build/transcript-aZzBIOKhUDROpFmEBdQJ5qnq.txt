+def _results_data():
+    """The Results page: one line per real book (backtest.json by_book), bound to the real trade counts,
+    win %, rupees and average R. No sample values — every number comes from the store, and the per-line
+    detail stays closed (open=False) so the sc-if trade rows are not required."""

+    except Exception:
+        books = []
+    tests = []
+    for b in books:
+        n = b.get("N") or 0
+        tot = b.get("total_rupees") or 0
+        avg = b.get("avg_R") or 0
+        if n == 0:
+            verdict, vcolor, won, made = "no trades yet", "#9a9a93", "—", "—"
+        elif tot > 0:
+            verdict, vcolor = "small plus — not proven yet, live paper decides", "#FAC775"
+            won = "%.1f%%" % (b.get("win_pct") or 0)

+            verdict, vcolor = "loses after costs", "#F09595"

+            "what": b.get("book", ""), "trades": str(n), "won": won, "made": made,
+            "verdict": verdict, "vColor": vcolor, "open": False, "toggle": "",
+        })
+    return {"tests": tests}
