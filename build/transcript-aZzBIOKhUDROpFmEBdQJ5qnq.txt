+"""R-NO-WINDOWS — one place that launches a desk process the right way: windowless (pythonw + CREATE_NO_WINDOW
+| DETACHED_PROCESS) with stdout AND stderr appended to a file under E1-DESK\\logs\\, never a console. Every
+launch point (recorders, runner, site_loop, watchdog relaunches, the post-login self-check, the supervisor and

+top-level windows and returns any owned by a desk python process — after a clean restart it must be empty."""
+from __future__ import annotations
+
+import os
+import subprocess
+import sys
+
+_REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+LOGDIR = os.path.join(os.path.expanduser("~"), "Desktop", "E1-DESK", "logs")

+CREATE_NO_WINDOW = 0x08000000
+DETACHED_PROCESS = 0x00000008
+DESK_PAT = ("scripts|engine1_rebuild|engine4_options|engine7_parity|engine9|engine12_xchange|"
+            "engine21_overnight|engine_youtube|book20_runner|supervisor|kite_catcher")

+def pythonw():
+    """The ONE windowless interpreter (Python 3.10 pythonw), else the current interpreter's pythonw, else this."""
+    if os.path.isfile(PY310W):
+        return PY310W
+    cand = os.path.join(os.path.dirname(sys.executable), "pythonw.exe")
+    return cand if os.path.isfile(cand) else sys.executable
+
+
+def _logfiles(name):
+    os.makedirs(LOGDIR, exist_ok=True)
+    safe = "".join(c if c.isalnum() or c in "-_." else "_" for c in str(name))
+    out = open(os.path.join(LOGDIR, safe + ".out"), "a", encoding="utf-8", errors="replace")
+    err = open(os.path.join(LOGDIR, safe + ".err"), "a", encoding="utf-8", errors="replace")
+    return out, err
+
+
+def launch(args, name, cwd=None):
+    """Start a desk process windowless with output to E1-DESK\\logs\\<name>.out/.err. `args` is the arg list
+    after the interpreter (e.g. ['-m','engine1_rebuild.book20_runner','--loop'] or ['scripts\\\\watchdog.py']).
+    Returns the Popen, or None on failure (never raises — a launch must never wedge the caller)."""
+    try:
+        out, err = _logfiles(name)
+        return subprocess.Popen([pythonw(), *args], cwd=cwd or _REPO, stdout=out, stderr=err,

+    except Exception:
+        return None
+
+
+def no_console_windows():

+    ps = ("Get-CimInstance Win32_Process | Where-Object { $_.Name -match 'python' -and "

+                             timeout=30, creationflags=CREATE_NO_WINDOW).stdout
+        return [int(x) for x in out.split() if x.strip().isdigit()]
+    except Exception:
+        return []

+"""R-ETA — a machine-measured schedule for the checklist. NOTHING here is a typed or guessed duration.
+
+- Item start/finish come from reports/ITEM-TIMES.json (stamped by build_journal.set_state on every item change).

+

+    """[(item, minutes)] for items with both a start and a finish."""

+                if mins > 0:

+                             capture_output=True, text=True, timeout=20).stdout
