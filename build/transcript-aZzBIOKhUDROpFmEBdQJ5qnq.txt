226	    """Write ONE name at Windows USER scope -- HKCU\\Environment, where setx writes.
227	
228	    The value is a parameter and a parameter only: it is not returned, not logged,
229	    not interpolated into any message, and not included in any exception this raises.
230	    REG_SZ unconditionally, so a key containing '%' stays literal and the value is

235	    with winreg.OpenKey(winreg.HKEY_CURRENT_USER, "Environment", 0,
236	                        winreg.KEY_SET_VALUE) as key:
237	        winreg.SetValueEx(key, name, 0, winreg.REG_SZ, value)

284	def lan_ip():
285	    """This laptop's address on the Wi-Fi, or None if it cannot be determined.
286	
287	    The UDP 'connect' sends no packet; it only asks the routing table which local
288	    address would be used to reach the outside, which is the one a phone on the same
289	    network can dial.
290	    """
291	    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
292	    try:
293	        sock.connect(("8.8.8.8", 80))
294	        return sock.getsockname()[0]
295	    except OSError:
296	        try:
297	            return socket.gethostbyname(socket.gethostname())
298	        except OSError:
299	            return None
300	    finally:
301	        sock.close()

306	class _Gate:
307	    """The mutable bits one run needs, kept off module globals so a test can hold two."""
308	
309	    def __init__(self, token, writer=None, render=None, parse=None, commit=None):
310	        self.token = [redacted]
311	        self.latch = OneShot()
312	        self.writer = writer or set_user_env
313	        self.done = threading.Event()
314	        self.lengths = None                        # what `commit` returned; None = nothing saved
315	        self._lock = threading.Lock()
316	        self.deadline = None
317	        self.hard_stop = None
318	        # THE THREE HOOKS (KITEDATA-1, 2026-08-19). A second gate exists now -- the daily
319	        # Kite TOKEN ceremony (`scripts/token_gate.py`) -- and it differs from this one in
320	        # exactly three places: what the page says, how a submission is read, and what is
321	        # done with it. Everything else it needs is the part of this file that was hard to
322	        # get right (one route, one shot, silence, no-store, the idle clock, the finally
323	        # that ends the run on the WRITE and not on the courtesy page). Those are hooks
324	        # rather than a copied handler because a duplicated credential handler is two
325	        # places to fix the next time one of them is measured wrong.
326	        self.render = render or page
327	        self.parse = parse or parse_submission
328	        self.commit = commit or self._write_both
329	
330	    def _write_both(self, values):
331	        """The key gate's own save: two USER-scope names, and the LENGTHS as its receipt."""
332	        for name, value in zip(KEY_NAMES, values):
333	            self.writer(name, value)
334	        return tuple(len(v) for v in values)
335	
336	    def open_at(self, now):
337	        self.hard_stop = now + HARD_CAP_MINUTES * 60
338	        self.touch(now)
339	
340	    def touch(self, now):
341	        """A real page load restarts the idle clock, never past the hard ceiling."""
342	        with self._lock:
343	            self.deadline = min(now + GATE_MINUTES * 60, self.hard_stop)
344	
345	    def expiry(self):
346	        with self._lock:
347	            return self.deadline

350	def _handler_for(gate):
351	    class Handler(http.server.BaseHTTPRequestHandler):
352	        server_version = "gate"
353	        sys_version = ""
354	        protocol_version = "HTTP/1.1"
355	
356	        # SILENCE. The stock handler prints every request line to stderr, token and
357	        # all; a console scrollback is exactly the kind of copy this file exists to
358	        # avoid making.
359	        def log_message(self, *_a):
360	            return
361	
362	        def _send(self, code, payload):
363	            self.send_response(code)
364	            self.send_header("Content-Type", "text/html; charset=utf-8")
365	            self.send_header("Content-Length", str(len(payload)))
366	            self.send_header("Cache-Control", "no-store")
367	            self.send_header("Referrer-Policy", "no-referrer")
368	            self.end_headers()
369	            self.wfile.write(payload)
370	
371	        def _reject(self):
372	            self._send(404, b"<!doctype html><title>404</title><h1>Not found</h1>")
373	
374	        def do_GET(self):                          # noqa: N802 - stdlib name
375	            if not path_matches(gate.token, self.path):
376	                return self._reject()
377	            if gate.latch.taken:
378	                return self._send(200, gate.render(gate.token, saved=True))
379	            gate.touch(time.monotonic())
380	            self._send(200, gate.render(gate.token))
381	
382	        def do_POST(self):                         # noqa: N802 - stdlib name
383	            if not path_matches(gate.token, self.path):
384	                return self._reject()
385	            try:
386	                length = int(self.headers.get("Content-Length") or 0)
387	            except ValueError:
388	                length = -1
389	            if length < 0 or length > MAX_BODY:
390	                return self._send(413, gate.render(gate.token,
391	                                            error="that was too big to be a submission"))
392	
393	            raw = self.rfile.read(length).decode("utf-8", "replace")
394	            values, reason = gate.parse(raw)
395	            del raw                                # the body has no second use
396	
397	            if values is None:
398	                # A fat-fingered submit must NOT burn the one shot.
399	                return self._send(400, gate.render(gate.token, error=reason))
400	            if not gate.latch.claim():
401	                return self._send(200, gate.render(gate.token, saved=True))
402	
403	            try:
404	                receipt = gate.commit(values)
405	            except Exception:                      # noqa: BLE001 - never echo the cause
406	                gate.lengths = None
407	                try:
408	                    self._send(500, gate.render(gate.token,
409	                                         error="this laptop refused the write - nothing was saved"))
410	                finally:
411	                    gate.done.set()                # see the note on the success path
412	                return
413	            gate.lengths = receipt
414	            del values
415	            broadcast_env_change()
416	            try:
417	                self._send(200, gate.render(gate.token, saved=True))
418	            finally:
419	                # THE WRITE ENDS THE RUN. THE COURTESY PAGE DOES NOT (measured
420	                # 2026-08-19, RESTART-4). The owner's phone reset this connection
421	                # while the reply was going out. The registry write had already
422	                # succeeded -- but `ConnectionResetError` came out of `_send` and
423	                # carried `done.set()` away with it, so the listener stayed open on
424	                # a saved gate, and the idle deadline was twenty minutes from
425	                # telling the owner it had "expired unused" when in fact it held
426	                # his keys. A `finally` costs one line; the alternative cost is a
427	                # door left open and a false sentence sent to the person who closed
428	                # it.
429	                gate.done.set()                    # the run ends; the server is torn down
430	
431	    return Handler

467	    """True only when stdout is a real interactive terminal.
468	
469	    WHY THIS EXISTS (KITEDATA-1's own hostile checker, 2026-08-19 — a leak this file
470	    opened and its 42 tests did not cover). `say` prints UNCONDITIONALLY, and it is not
471	    a Telegram-failure fallback: the open line always reaches stdout. That was safe
472	    while the only caller was a human double-clicking a `.bat`. It stopped being safe
473	    the moment a CHAINED run could open a gate, because `scheduler/launch_gate.py`

482	    sees it exactly as before, and a headless run says nothing it cannot unsay. The

484	    expires unused and says so, which is strictly better than a published door.

539	def _notify_url(text):
540	    """The open line -- Telegram or NOTHING. The URL never touches the disk.
541	
542	    WHY THIS DOES NOT REUSE `scheduler/ping.py` (measured 2026-08-19: that module was
543	    falling back with `http 502` on this box at the time this ran). Its fallback
544	    writes the message into `docs/STATUS_FEED.md`, a tracked file that gets pushed --
545	    which would publish a live gate URL to anyone with the repo for as long as the
546	    window lasts. The token is not a credential, but a door that is open for twenty
547	    minutes should not be advertised in git. So the failure mode here is deliberately
548	    "the owner reads it off this console", never "it is written down".
549	
550	    Returns True if Telegram accepted it. Never raises. Never echoes a credential:
551	    an HTTPError's message can carry the request URL, which carries the bot token, so
552	    only the status CODE is ever reported.
553	    """
554	    token = [redacted]("TELEGRAM_BOT_TOKEN")
555	    chat = os.environ.get("TELEGRAM_CHAT_ID")
556	    if not token or not chat:
557	        say("telegram not configured - " + _undelivered())
558	        return False
559	    import json
560	    import urllib.error
561	    import urllib.request
562	    payload = json.dumps({"chat_id": [redacted], "text": text,
563	                          "disable_web_page_preview": True}).encode("utf-8")
564	    req = urllib.request.Request(
565	        "https://api.telegram.org/bot{}/sendMessage".format(token), data=payload,
566	        headers={"Content-Type": "application/json"}, method="POST")
567	    try:
568	        with urllib.request.urlopen(req, timeout=15) as resp:
569	            if resp.status == 200:
570	                return True
571	            say("telegram refused (http {}) - {}".format(resp.status, _undelivered()))
572	    except urllib.error.HTTPError as exc:
573	        say("telegram refused (http {}) - {}".format(exc.code, _undelivered()))
574	    except Exception as exc:                       # noqa: BLE001 - never fatal
575	        say("telegram unreachable ({}) - {}".format(type(exc).__name__, _undelivered()))
576	    return False

579	def serve(gate, notes, notify=None, notify_url=None, port=0):
580	    """Bind, self-check, announce, wait for ONE save or the deadline, close. Exit code.
581	
582	    0 = saved, 2 = expired unused, 3 = it could not be opened honestly,
583	    4 = the submission arrived but this laptop refused the write.
584	
585	    EXTRACTED FROM `run` BY KITEDATA-1 so the second gate (`scripts/token_gate.py`)
586	    inherits this body instead of copying it. `notes` supplies the four sentences that
587	    differ between gates: `open(ip, port, token)`, `saved(receipt)`, `expired()`,
588	    `refused()`. `port` is 0 (a random one) everywhere except the laptop route, where
589	    the gate must answer on the port the Kite redirect actually points at.
590	
591	    TWO notifiers on purpose: `notify` may fall back to the tracked status feed,
592	    `notify_url` may not. The URL line is the only one that must never be written down.
593	    """
594	    notify = notify or _default_notify
595	    notify_url = notify_url or _notify_url
596	    server = http.server.ThreadingHTTPServer(("0.0.0.0", port), _handler_for(gate))
597	    server.daemon_threads = True
598	    bound = server.server_address[1]
599	    gate.open_at(time.monotonic())
600	
601	    threading.Thread(target=server.serve_forever, daemon=True).start()
602	    try:
603	        if not _self_check(bound, gate.token):
604	            notify("[TRADING] Gate could not answer on its own port - not opened.")
605	            say("self-check failed - gate not opened")
606	            return 3
607	        ip = lan_ip()
608	        if ip is None:
609	            notify("[TRADING] Gate could not find this laptop's Wi-Fi address - not opened.")
610	            say("no LAN address - gate not opened")
611	            return 3
612	        line = notes["open"](ip, bound, gate.token)
613	        # THE URL IS SAID TO A CONSOLE ONLY WHEN A HUMAN IS AT ONE (see
614	        # `a_human_is_watching`): a headless run's stdout is streamed into
615	        # `reports/live_run.log` and force-pushed to the `live` branch.
616	        say_url(line, "gate open - its URL was not written to this console")
617	        notify_url(line)
618	
619	        while not gate.done.wait(timeout=1.0):
620	            if time.monotonic() >= gate.expiry():
621	                notify(notes["expired"]())
622	                say("gate expired unused - nothing saved")
623	                return 2
624	        if gate.lengths is None:
625	            notify(notes["refused"]())
626	            say("write refused - nothing saved")
627	            return 4
628	        notify(notes["saved"](gate.lengths))
629	        say("saved - gate closed")
630	        return 0
631	    finally:
632	        server.shutdown()
633	        server.server_close()

636	def run(notify=None, notify_url=None, writer=None, already_set=None):
637	    """The KEY gate: refuse if the keys are already set, else `serve` one submission."""
638	    notify = notify or _default_notify
639	    notify_url = notify_url or _notify_url
640	    # THE REFUSAL COMES FIRST, BEFORE A SOCKET EXISTS (dispatch UNPARK-2 step E). A
641	    # second gate over keys that are already set cannot help and can overwrite; the
642	    # only safe version of it is the one that never binds.
643	    if (already_set or keys_already_set)():
644	        notify(already_set_note())
645	        say("keys already set, gate not needed")
646	        return 5
647	    return serve(
648	        _Gate(make_token(), writer=writer),
649	        {"open": open_note,
650	         "saved": lambda receipt: presence_note(*receipt),
651	         "expired": expired_note,
652	         "refused": lambda: "[TRADING] Key gate: this laptop refused the write - nothing saved."},
653	        notify=notify, notify_url=notify_url)

313	## R15 — A control file that fails validation is treated as ABSENT, never as permissive
314	
315	A **control file** is any file one component writes so another component may act on it: a cut factor,
316	a regime label, an allocation, a gate verdict, a freshness stamp. Where such a file is missing, a
317	system fails closed. This ruling says a **corrupt** one must fail the same way.
318	
319	| Part | Ruling |
320	|---|---|
321	| **Metric** | Whether a read control file passes its declared validation — every required row present, every value inside its declared range, the file readable and stamped with the cycle and version it claims. |

323	| **Threshold** | **Any single validation failure** — one out-of-range value, one missing row, one unreadable or unstamped file. There is no partial-trust reading. |
