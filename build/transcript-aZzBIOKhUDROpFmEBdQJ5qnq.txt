+from __future__ import annotations
+

+def _scrub(text):
+    try:
+        from scripts import build_journal
+        return build_journal.scrub(str(text))
+    except Exception:
+        return str(text)
+
+
+def _write(direction, text):
+    try:
+        line = "%s\t%s\t%s" % (datetime.now(IST).isoformat(), direction,
+                               _scrub(text).replace("\t", " ").replace("\n", " ")[:1000])
+        os.makedirs(os.path.dirname(MSGLOG), exist_ok=True)
+        with open(MSGLOG, "a", encoding="utf-8") as f:
+            f.write(line + "\n")
+    except Exception:
+        pass
+
+
+def outgoing(text):
+    _write("OUT", text)
+
+
+def incoming(text):
+    _write("IN", text)
+
+
+def recent(n=400):
+    """[(iso, direction, text)] newest first."""
+    try:
+        rows = [l.split("\t", 2) for l in open(MSGLOG, encoding="utf-8").read().splitlines() if l.strip()]
+        rows = [r for r in rows if len(r) == 3]
+        return rows[::-1][:n]
+    except Exception:
+        return []
