+"""R-EVENING-22 C2 — adaptive sizing EXACTLY as the math dictates, on ₹2,00,000, for the certified survivors.
+
+It runs the built adaptive sizer (engine1_rebuild.adaptive_sizing.size — Wilson-LB accuracy × win/loss at
+¼-Kelly on current equity, bounded by the pattern's drawdown, a total-risk cap and a cluster cap, fail-closed
+to flat when a measurement is missing). NO size is typed — every rupee is the sizer's own output, and the
+reason it chose that size is printed beside it.
+
+On today's five survivors the sizer stays FLAT: each has 9–13 fires (below the 30-fire floor) and no measured
+book drawdown yet, which is precisely the confidence-adjusted, fail-closed behaviour — adaptive is never
+larger than the ₹1,000 reference on a number it cannot support. That is the honest C2 result, not a bug."""
+from __future__ import annotations
+
+import json
+import os
+import sys
+
+sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
+from engine1_rebuild import datastore as ds
+
+REPO = ds._REPO
+EQUITY = 200000
+OUT_JSON = os.path.join(REPO, "reports", "C2-SIZING.json")
+OUT_MD = os.path.join(REPO, "reports", "C2-SIZING.md")
+
+
+def _survivors():
+    try:
+        d = json.load(open(os.path.join(REPO, "reports", "EXAMS-REAL-STAGING.json"), encoding="utf-8"))
+        return d.get("certified_patterns") or []
+    except Exception:
+        return []
+
+
+def size_all(equity=EQUITY):
+    from engine1_rebuild import adaptive_sizing as AS
+    rows = []
+    for c in _survivors():
+        scorecard = {
+            "fires": c.get("fires"),
+            "win_rate": (c.get("win_pct") or 0) / 100.0,      # measured hit rate
+            "win_R": abs(float(c.get("mean_net_R") or 1.5)) or 1.5,
+            "loss_R": 1.0,                                     # 1R = the stop distance by construction
+        }
+        # worst_daily_dd_rupees is the BOOK's measured worst daily drawdown — not measured per pattern yet,
+        # so it is left None: the sizer then fails closed to flat, which is the correct confidence-adjusted
+        # behaviour on a thin, young pattern. Nothing here is a typed size.
+        sized = AS.size(scorecard, equity, worst_daily_dd_rupees=None)
+        rows.append({"stock": c.get("stock"), "family": c.get("family"), "fires": c.get("fires"),
+                     "win_pct": c.get("win_pct"), "risk_rupees": sized.get("risk_rupees"),
+                     "risk_fraction": sized.get("risk_fraction"), "mode": sized.get("mode"),
+                     "reason": sized.get("reason"), "frac_kelly": sized.get("frac_kelly")})
+    return rows
+
+
+def run(write=True, equity=EQUITY):
+    rows = size_all(equity)
+    adaptive = [r for r in rows if r["mode"] == "adaptive"]
+    out = {"equity": equity, "survivors": rows, "adaptive_count": len(adaptive), "flat_count": len(rows) - len(adaptive),
+           "note": ("Every rupee is the sizer's own output (no typed sizes). The survivors size FLAT because "
+                    "each is below the 30-fire floor with no measured book drawdown — adaptive never exceeds "
+                    "the flat reference on a number it cannot support (fail-closed).")}
+    if write:
+        with open(OUT_JSON, "w", encoding="utf-8") as f:
+            json.dump(out, f, indent=1, default=str)
+        _write_md(out)
+    return out
+
+
+def _rup(x):
+    try:
+        return "₹%s" % format(int(round(float(x))), ",")
+    except (TypeError, ValueError):
+        return "—"
+
+
+def _write_md(o):
+    L = ["# C2 — ADAPTIVE SIZING ON ₹2,00,000 (the survivors)", "",
+         "_The built adaptive sizer (¼-Kelly on the Wilson-LB accuracy, bounded by pattern drawdown, a "
+         "total-risk cap and a cluster cap, fail-closed to flat). No size is typed — every rupee is the "
+         "sizer's output, with the reason it chose it._", "",
+         "| Stock | Pattern | Fires | Win %% | Size chosen | Mode | Why |",
+         "|---|---|---|---|---|---|---|"]
+    for r in o["survivors"]:
+        L.append("| %s | %s | %s | %s | %s | %s | %s |" % (
+            r["stock"], r["family"], r["fires"], r["win_pct"], _rup(r["risk_rupees"]), r["mode"], r["reason"]))
+    L += ["", "**%d of %d sized adaptively; %d stayed flat.** The flat ones are correct: below the 30-fire "
+          "floor with no measured book drawdown, the sizer will not exceed the ₹1,000 reference. Adaptive "
+          "sizing turns on for a pattern the moment it has ≥30 fires and a measured drawdown."
+          % (o["adaptive_count"], len(o["survivors"]), o["flat_count"])]
+    with open(OUT_MD, "w", encoding="utf-8") as f:
+        f.write(("\n".join(L) + "\n").replace("Win %%", "Win %"))
+
+
+def _demo():
+    o = run(write=False)
+    assert "survivors" in o and o["adaptive_count"] + o["flat_count"] == len(o["survivors"])
+    # every size is the sizer's own output with a stated reason — never a typed number
+    for r in o["survivors"]:
+        assert r["risk_rupees"] is not None and r["reason"]
+    print("c2_sizing demo OK: %d survivors sized (%d adaptive, %d flat), no typed sizes." % (
+        len(o["survivors"]), o["adaptive_count"], o["flat_count"]))
+
+
+if __name__ == "__main__":
+    if "--demo" in sys.argv:
+        _demo()
+    else:
+        o = run()
+        print("C2 sizing: %d survivors — %d adaptive, %d flat (see reports/C2-SIZING.md)." % (
+            len(o["survivors"]), o["adaptive_count"], o["flat_count"]))
