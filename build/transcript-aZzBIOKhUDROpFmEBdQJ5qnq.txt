+
+## Root cause of the two failures
+
+1. **`codex login` → socket error 10013 (WSAEACCES).** The OAuth loopback callback port codex uses,
+   **1455, sits inside the Windows excluded TCP range 1373–1472** (part of the 1073–1472 block reserved by
+   Windows NAT / Hyper-V / WSL). A process cannot bind a port inside an excluded range, so the callback
+   server bind fails with 10013. **This is the answer to the dispatch's direct question: yes, 1455 is inside
+   an excluded range.**
+
+2. **`codex login --device-auth` → "error sending request".** The device-auth path does **not** use a
+   loopback port, so it is not affected by (1). When I ran `codex login --device-auth` here it did **not**
+   error out — it ran for 45s+ with no "error sending request", which indicates egress from codex to
+   auth.openai.com **works** on this laptop now (the earlier failure was likely transient or from a
+   sandboxed context). **But** device-auth needs a real terminal (TTY) to display its link + one-time code;
+   run head-less through the agent's piped Bash it emits **nothing** (0 bytes captured, still "Not logged
+   in"), so I cannot read or relay the code from here.
+
+## Fix options
+
+- **No-admin, one link (recommended).** The owner runs **`codex login --device-auth`** in their **own
+  PowerShell** window. It prints a verification link + a short code; the owner opens the link and enters the
+  code. No admin rights, no port fix needed — egress is not the blocker. *(I cannot drive this for the owner
+  because the code only renders on a real terminal, not the agent's pipe.)*
+
+- **Admin, to make the normal `codex login` loopback flow work.** Free port 1455 from the excluded range.
+  The 1073–1472 block is a dynamic winnat reservation; in an **elevated** PowerShell:
+  `net stop winnat` then `net start winnat` (regenerates the dynamic ranges, usually off the low ports),
+  or explicitly reserve the flow's range away from 1455. This needs admin and is **not** required if the
+  device-auth path above is used.
+
+## Verdict
+
+Sign-in is **not network-blocked** and needs **no admin** — it needs a real terminal the agent doesn't
+have. The single owner action that finishes it is one command in their own PowerShell plus one link tap.
+`codex exec "reply with the word ok"` (step 4) can only be run **after** that sign-in completes.

+def _stocks_data():
+    """The Stocks page: every stock the desk watches, from the real screener (site/data/screener.json, 210
+    rows). Real columns bind — name, volume (rvol), trend — and the readings the screener does not carry
+    (RSI/MFI/price-range percentile/5-20d returns/gap) show an honest '—' rather than a fabricated
+    number (§4.4). 'Signal today' is cross-referenced from the real calls (live.json), not invented; a
+    stock is 'ready' when its screener row is not stale. The design's 30 procedurally-generated stocks are
+    dropped."""
+    import json
+    def _load(path, key=None):
+        try:
+            d = json.load(open(os.path.join(REPO, path), encoding="utf-8"))
+            return d.get(key) if key else d
+        except Exception:
+            return None
+    rows_src = _load("site/data/screener.json", "rows") or []
+    sig_stocks = {(c.get("stock") or "").upper() for c in (_load("site/data/live.json", "calls") or [])}
+    DASH, GREY, GREEN = "—", "#9a9a93", "#5DCAA5"
+    out = []
+    for r in rows_src:
+        sym = (r.get("symbol") or "").upper()
+        sig = sym in sig_stocks
+        rvol = r.get("rvol")
+        out.append({
+            "name": sym, "range": DASH, "rc": GREY,
+            "mom": DASH, "calm": DASH, "flow": DASH,
+            "vol": ("%.1fx" % rvol) if isinstance(rvol, (int, float)) else DASH,
+            "gap": DASH,
+            "met": "signal" if sig else DASH, "mc": GREEN if sig else GREY,
+            "m5": DASH, "m20": DASH, "gapd": DASH,
+            "trend": r.get("trend") or DASH, "r60": DASH,
+            "sig": sig, "ready": not r.get("stale", False),
+        })
+    out.sort(key=lambda x: (not x["sig"], x["name"]))     # real signals first, then alphabetical
+    ready_n = sum(1 for x in out if x["ready"])
+    on, off, onFg, offFg = "#5DCAA5", "#161920", "#0f1115", "#b8b6ae"
+    return {
+        "rows": out, "count": len(out), "note": "signals first, then the rest of the watch list",
+        "allBg": on, "allFg": onFg, "sigBg": off, "sigFg": offFg, "readyBg": off, "readyFg": offFg,
+        "pickAll": "", "pickSig": "", "pickReady": "", "typeQuery": "",
+        "_ready_n": ready_n, "_total": len(out),   # consumed by _fix_stocks, stripped before render
+    }

+def _fix_stocks(html):
+    """Owner title 'Stocks the desk watches — N' and the real ready-count on the button (the design bakes

+    d = _stocks_data()

+    html = html.replace("Ready to trade (84)", "Ready to trade (%d)" % ready_n)
+    return html

+The drop folder is `Desktop\\E1-DESK\\drops`. A drop is any file placed there. Each new file is parsed for a

+fire exactly once per file. Nothing here touches the trading path.
+
+DATA-AS-INSTRUCTIONS (§20.7): a dropped page is external DATA. This reader extracts fields (title, first
+readable line) and NOTHING else — imperative text inside a drop is inert and is never executed, followed, or
+allowed to change a rule. The extracted text is scrubbed before it reaches Telegram or the log.

+import json
+import os
+import re

+REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+DROPS_DIR = os.path.join(os.path.dirname(REPO), "E1-DESK", "drops")
+SEEN = os.path.join(REPO, "scheduler", ".drops_seen.json")
+LOG = os.path.join(REPO, "reports", "ANALYST-DROPS.md")
+

+def _load_seen():
+    try:
+        return set(json.load(open(SEEN, encoding="utf-8")))
+    except Exception:
+        return set()
+
+
+def _save_seen(seen):
+    try:
+        os.makedirs(os.path.dirname(SEEN), exist_ok=True)
+        json.dump(sorted(seen), open(SEEN, "w", encoding="utf-8"))
+    except Exception:
+        pass
+
+
+def _strip_html(text):
+    text = re.sub(r"(?is)<(script|style)\b.*?</\1>", " ", text)
+    text = re.sub(r"(?s)<[^>]+>", " ", text)
+    text = re.sub(r"&[a-z#0-9]+;", " ", text)
+    return re.sub(r"\s+", " ", text).strip()
+
+
+def parse(path):
+    """Return (title, one_line_summary) from a dropped file — fields only, never executed content."""
+    name = os.path.basename(path)
+    try:
+        raw = open(path, encoding="utf-8", errors="replace").read()
+    except Exception:
+        return name, ""
+    title = ""
+    m = re.search(r"(?is)<title>\s*(.*?)\s*</title>", raw)
+    if m:
+        title = _strip_html(m.group(1))[:120]
+    body = _strip_html(raw) if ("<" in raw and ">" in raw) else re.sub(r"\s+", " ", raw).strip()
+    if not title:
+        title = (body[:80] or name)
+    summary = body[:200]
+    return title, summary
+
+
+def scan(post_fn=None):
+    """Pick up any NEW file in the drops folder: parse, log, post ONE scrubbed line. Returns the list of
+    filenames newly handled this call. Fires once per file (the seen-list). No folder = nothing to do."""
+    from scripts import build_journal   # reuse the one shared secret scrubber
+    if post_fn is None:
+        from scripts import notify
+        post_fn = notify.notify
+    if not os.path.isdir(DROPS_DIR):
+        return []
+    seen = _load_seen()
+    handled = []
+    for name in sorted(os.listdir(DROPS_DIR)):
+        full = os.path.join(DROPS_DIR, name)
+        if name in seen or not os.path.isfile(full):
+            continue
+        title, summary = parse(full)
+        line = build_journal.scrub("Analyst drop: %s - %s" % (name, title))[:280]
+        try:
+            with open(LOG, "a", encoding="utf-8") as f:
+                f.write("- %s -> %s\n" % (name, build_journal.scrub(summary)[:200]))
+        except Exception:
+            pass
+        try:
+            post_fn(line)
+        except Exception:
+            pass
+        seen.add(name)
+        handled.append(name)
+    if handled:
+        _save_seen(seen)
+    return handled
+
+
+def tick():
+    """Periodic entry, called from token_ping's loop like build_beat/kill_switch. Never fatal."""
+    try:
+        return scan()
+    except Exception:
+        return []

+    _demo()

+"""Weekly live-cost check — the real cost charged per paper trade vs the calibrated backtest cost basis
+(BACKLOG item, owner ruling R-BACKLOG, 2026-09-22). One plain Telegram line, once per ISO week.
+
+The two sides:
+  • LIVE — the real `cost_R` recorded on each paper trade in engine1_rebuild/paper/ledger.csv.
+  • CALIBRATED BASIS — money.cost_readout()['cost_round_trip'] (statutory + 2*half_spread), the same cost the
+    backtest was settled under. While the half-spread is COLD-START (no measured depth yet), that basis IS the
+    conservative fallback, so the delta is ~0 by construction until real fills accumulate — the badge says so.
+
+Badges (§19): COLD-START while the spread is the cited fallback, FILL-LAW-UNVERIFIED until the shadow re-settle
+delta is published (D1). Nothing here touches the trading path; a missing ledger or cost model is a safe no-op.
+"""
+from __future__ import annotations
+
+import csv
+import os
+
+REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+LEDGER = os.path.join(REPO, "engine1_rebuild", "paper", "ledger.csv")
+LOG = os.path.join(REPO, "reports", "COST-LEDGER.md")
+WEEK_SLOT = os.path.join(REPO, "scheduler", ".cost_week")     # gitignored: last ISO week posted
+
+
+def _rows():
+    try:
+        return list(csv.DictReader(open(LEDGER, encoding="utf-8")))
+    except Exception:
+        return []
+
+
+def _fnum(v):
+    try:
+        return float(v)
+    except (TypeError, ValueError):
+        return None
+
+
+def _basis():
+    """(assumed round-trip cost in R, is_cold_start). The cost model reports COST in fraction-of-capital terms;
+    1R = money.PAPER_R_RUPEES, so cost_R = cost_fraction * NOTIONAL_CAPITAL / PAPER_R_RUPEES."""
+    try:
+        from engine1_rebuild import money
+        ro = money.cost_readout()
+        cost_frac = ro["cost_round_trip"]["value"]                    # fraction of notional, round trip
+        cold = "COLD-START" in (ro["half_spread"].get("source") or "")
+        cost_r = cost_frac * money.NOTIONAL_CAPITAL / money.PAPER_R_RUPEES
+        return round(cost_r, 4), cold
+    except Exception:
+        return None, True
+
+
+def weekly_line():
+    """One plain line comparing the real per-trade cost to the calibrated basis. Honest empty-state when no
+    live round-trips have closed yet."""
+    rows = _rows()
+    costs = [_fnum(r.get("cost_R")) for r in rows]
+    costs = [c for c in costs if c is not None]
+    closed = [r for r in rows if r.get("exit_date") and _fnum(r.get("cost_R")) is not None]
+    basis, cold = _basis()
+    badge = ("[COLD-START]" if cold else "") + "[FILL-LAW-UNVERIFIED]"
+    basis_s = ("%.3fR" % basis) if basis is not None else "n/a"
+    if not costs:
+        return "Cost check: no paper trades carry a cost yet - basis %s. %s" % (basis_s, badge)
+    live_avg = sum(costs) / len(costs)
+    if closed:
+        cl = [_fnum(r.get("cost_R")) for r in closed]
+        live_closed = sum(cl) / len(cl)
+        delta = (" - delta %+.3fR vs basis" % (live_closed - basis)) if basis is not None else ""
+        return "Cost check: %d live round-trips, real cost avg %.3fR vs calibrated %s%s. %s" % (
+            len(closed), live_closed, basis_s, delta, badge)
+    return ("Cost check: %d open positions, entry cost avg %.3fR; basis %s; 0 round-trips closed yet "
+            "- cost-vs-backtest delta lands when trades close. %s") % (len(costs), live_avg, basis_s, badge)
+
+
+def _iso_week():
+    import datetime
+    y, w, _ = datetime.date.today().isocalendar()
+    return "%d-W%02d" % (y, w)
+
+
+def maybe_post(post_fn=None, now_week=None):
+    """Post the weekly line at most once per ISO week. Returns the line if posted, else None."""
+    week = now_week or _iso_week()
+    try:
+        last = open(WEEK_SLOT, encoding="utf-8").read().strip()
+    except Exception:
+        last = ""
+    if last == week:
+        return None
+    line = weekly_line()
+    if post_fn is None:
+        from scripts import notify
+        post_fn = notify.notify
+    try:
+        post_fn(line)
+    except Exception:
+        pass
+    try:
+        with open(LOG, "a", encoding="utf-8") as f:
+            f.write("- %s: %s\n" % (week, line))
+    except Exception:
+        pass
+    try:
+        os.makedirs(os.path.dirname(WEEK_SLOT), exist_ok=True)
+        open(WEEK_SLOT, "w", encoding="utf-8").write(week)
+    except Exception:
+        pass
+    return line
+
+
+def tick():
+    """Periodic entry from token_ping's loop; never fatal."""
+    try:
+        return maybe_post()
+    except Exception:
+        return None
+
+
+def _demo():
+    line = weekly_line()
+    assert "Cost check" in line and "FILL-LAW-UNVERIFIED" in line, line
+    posts = []
+    a = maybe_post(post_fn=posts.append, now_week="1999-W01")
+    b = maybe_post(post_fn=posts.append, now_week="1999-W01")   # same week -> no second post
+    assert a is not None and b is None and len(posts) == 1, (a, b, posts)
+    print("cost_ledger demo OK: weekly line built, posts once per ISO week ::", line)
+
+
+if __name__ == "__main__":
+    _demo()
